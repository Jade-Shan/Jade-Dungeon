# 修复 map.js 第 134 行 ERR_HTTP_HEADERS_SENT 报错

## Context

`parseImage` handler（行 95-167）作为图片代理，请求外部图片并将其字节流返回给客户端。响应写入全部发生在 `res.on('close', ...)` 事件回调中（异步触发），但 handler 函数中包含一个 **永远不会 resolve 的 Promise**（`getImgResp`），导致 handler 永远无法正常返回，框架也无法执行默认的 header 写入逻辑。然而一旦任何外部因素（如超时中间件、反向代理断开连接、或未来修复了 Promise）导致框架在 `close` 事件之前先行发送了响应头，`close` 回调中的 `writeHead` 就会触发 `ERR_HTTP_HEADERS_SENT`。

## 根因

- `getImgResp` Promise 的 `resolve`/`reject` 从未被调用，Promise 永久 pending
- 代理图片响应的写入全在 `res.on('close', ...)` 异步事件中完成
- handler 函数与框架的自动响应写入机制之间存在竞争条件

## 修改方案

将图片代理逻辑封装为一个**可正确 resolve 的 Promise**，handler `await` 该 Promise 完成后再返回控制权给框架。这样：
- Promise resolve 时，`close` 回调已经完成 `writeHead` + `end`
- handler 返回后，`headersSent` 已经是 `true`，框架跳过默认 header 写入
- 不再有竞争条件

```javascript
"/api/sandtable/parseImage": async (context, data) => {
    let imgUrl = data.params.src;
    console.log(imgUrl);

    // 用 Promise 封装整个图片请求+响应周期，确保 handler 不提前返回
    await new Promise((resolve, reject) => {
        let handle = (res) => {
            const buffers = [];
            res.on('data', (chunk) => { buffers.push(chunk); });
            res.on('end', () => {
                let body;
                try {
                    body = Buffer.concat(buffers);
                } catch (e) {
                    body = Buffer.from(`\n    ${e}`);
                }
                if (res.statusCode !== 200) {
                    console.error(`Did not get an OK from the server. Code: ${res.statusCode}`);
                    context.response.writeHead(500, {
                        'Content-Type': res.headers['content-type'] || 'text/plain',
                        'Cache-Control': 'public,s-maxage=300,max-age=300',
                        'Access-Control-Allow-Origin': '*',
                        'Access-Control-Allow-Methods': 'GET,POST',
                        'Access-Control-Allow-Headers': 'x-requested-with,content-type'
                    });
                    let errBody = 'Err on get : ' + imgUrl;
                    errBody = errBody + '\n   call src err-code: ' + res.statusCode;
                    try {
                        errBody = errBody + '\n    ' + body.toString();
                    } catch (e) {
                        errBody = errBody + '\n    ' + e;
                    }
                    context.response.end(errBody);
                } else {
                    context.response.writeHead(200, {
                        'Content-Type': res.headers['content-type'] || 'application/octet-stream',
                        'Cache-Control': 'private,maxage=36000',
                        'Access-Control-Allow-Origin': '*',
                        'Access-Control-Allow-Methods': 'GET,POST',
                        'Access-Control-Allow-Headers': 'x-requested-with,content-type'
                    });
                    context.response.end(body);
                }
                resolve();
            });
            res.on('error', (err) => {
                console.error(`Response stream error for ${imgUrl}:`, err);
                if (!context.response.headersSent) {
                    context.response.writeHead(502, {
                        'Content-Type': 'application/json;charset=utf-8',
                        'Access-Control-Allow-Origin': '*',
                        'Access-Control-Allow-Methods': 'GET,POST',
                        'Access-Control-Allow-Headers': 'x-requested-with,content-type'
                    });
                    context.response.end(JSON.stringify({ status: "error", msg: "image fetch failed", data: imgUrl }));
                }
                resolve();
            });
        };

        let request;
        if (/^https:\/\//i.test(imgUrl)) {
            request = https.get(imgUrl,
                { agent: new https.Agent({ rejectUnauthorized: false, keepAlive: true }) },
                handle);
        } else {
            request = http.get(imgUrl, handle);
        }
        request.on('error', (err) => {
            console.error(`Request error for ${imgUrl}:`, err);
            if (!context.response.headersSent) {
                context.response.writeHead(502, {
                    'Content-Type': 'application/json;charset=utf-8',
                    'Access-Control-Allow-Origin': '*',
                    'Access-Control-Allow-Methods': 'GET,POST',
                    'Access-Control-Allow-Headers': 'x-requested-with,content-type'
                });
                context.response.end(JSON.stringify({ status: "error", msg: "image request failed", data: imgUrl }));
            }
            resolve();
        });
        request.setTimeout(15000, () => {
            request.destroy();
            if (!context.response.headersSent) {
                context.response.writeHead(504, {
                    'Content-Type': 'application/json;charset=utf-8',
                    'Access-Control-Allow-Origin': '*',
                    'Access-Control-Allow-Methods': 'GET,POST',
                    'Access-Control-Allow-Headers': 'x-requested-with,content-type'
                });
                context.response.end(JSON.stringify({ status: "error", msg: "image request timeout", data: imgUrl }));
            }
            resolve();
        });
    });
},
```

### 修改要点说明

| 要点 | 说明 |
|------|------|
| 非 HTTP URL 提前返回 | 在校验阶段就 `writeHead` + `end` 并 `return`，与框架机制兼容 |
| Promise 正确 resolve | Promise 在 `end` 事件回调中 `resolve()`，确保 `writeHead` + `end` 在 handler 返回前完成 |
| 监听 `end` 而非 `close` | `end` 在数据消费完毕后触发，语义更准确；`close` 可能在 `end` 之后触发且可能伴随错误状态 |
| 错误处理完整 | 分别处理：URL 协议校验 → HTTP 请求级错误（`request.on('error')`）→ 响应流错误（`res.on('error')`）→ 非 200 状态码 |
| 超时保护 | `request.setTimeout(15000)` 防止请求永远挂起 |
| 容错守卫 `headersSent` | 错误回调中检查 `headersSent`，防止重复写入响应 |

## 兼容性

- 修改范围仅限 `parseImage` 一个 handler
- 对其他 handler（`load-move-request`、`request-move`、`map-owner`、`load-map`、`save-map`）无影响
- `simpleHTTPServer.js` 框架逻辑无需变更 — handler 返回时 `headersSent` 为 true，框架自动跳过默认 header 写入
- 对外部调用者完全透明：请求格式和响应格式不变

# 全仓库同类问题排查

基于 `parseImage` 的修复模式，对全部 handler 和辅助函数进行了排查。核心排查维度：

| 维度 | 说明 |
|------|------|
| Promise 永不 resolve/reject | `new Promise(...)` 但 resolve/reject 在某些路径上不会被调用 |
| 异步响应写入竞争 | 响应写入发生在事件回调中，handler 可能在写入完成前返回 |
| 缺少 `request.on('error')` | HTTP 请求没有 error 事件处理，网络错误时 Promise 挂起 |
| 缺少超时保护 | 没有 `setTimeout`，慢速/挂起的连接会导致 Promise 永久 pending |
| `headersSent` 检查缺失 | 异常路径下未检查响应头是否已发送 |

## 问题一：`weatherapi.js` — `fetchForecastData` 缺少错误处理与超时

### 文件：`javascript/utils/weatherapi.js`，第 52-77 行

### 问题描述

`fetchForecastData` 中创建 Promise 封装 `https.get` 请求，但：
1. **没有 `request.on('error')`** — 网络错误（DNS 失败、连接被拒等）时 Promise 永不 settle
2. **没有超时保护** — 服务器响应但永不关闭连接时 Promise 永久 pending
3. 非 200 时 `reject()` 后未 `return`，继续注册 `data`/`close` 事件（逻辑不严谨）
4. 使用 `close` 事件而非 `end` 事件（`close` 可能在错误状态后触发）

该函数被 `exports.fetchForecast` 调用并被 `weather.js` handler `await`，一旦挂起，整个 `/api/weather/forecast` 接口将永久无响应。

### 修改前的代码

```javascript
let fetchForecastData = async (appKey, cityName, days) => {
    let result = { status: "err", msg: "unknow err" };
    let pms = new Promise((resolve, reject) => {
        const buffers = [];
        let body = null;
        let request = https.get(`https://api.weatherapi.com/v1/forecast.json?key=${appKey}&q=${cityName}&days=${days}&aqi=no&alerts=no`,
            (res) => {
                if (res.statusCode != 200) {
                    reject(`Server HTTP Err Code: ${res.statusCode}`);
                }
                res.on('data', (chunk) => { buffers.push(chunk); });
                res.on('close', () => {
                    let str = Buffer.concat(buffers).toString('utf8');
                    try {
                        resolve(JSON.parse(str));
                    } catch (e) { reject(`parse json err: ${str}`); }
                });
            });
    });
    await pms.then((data) => {
        result = { status: 'success', msg: '', oriData: data };
    }).catch((e) => {
        result.msg = e;
    });
    return result;
};
```

### 修改后的代码

```javascript
let fetchForecastData = async (appKey, cityName, days) => {
    let result = { status: "err", msg: "unknow err" };
    await new Promise((resolve, reject) => {
        const buffers = [];
        let settled = false;
        const done = (err, data) => {
            if (settled) return;
            settled = true;
            if (err) { reject(err); } else { resolve(data); }
        };

        let request = https.get(
            `https://api.weatherapi.com/v1/forecast.json?key=${appKey}&q=${cityName}&days=${days}&aqi=no&alerts=no`,
            (res) => {
                if (res.statusCode !== 200) {
                    // 消费响应体以防内存泄漏，然后 reject
                    res.resume();
                    return done(new Error(`Server HTTP Err Code: ${res.statusCode}`));
                }
                res.on('data', (chunk) => { buffers.push(chunk); });
                res.on('end', () => {
                    let str = Buffer.concat(buffers).toString('utf8');
                    try {
                        done(null, JSON.parse(str));
                    } catch (e) {
                        done(new Error(`parse json err: ${str}`));
                    }
                });
                res.on('error', (err) => {
                    done(new Error(`Response stream error: ${err.message}`));
                });
            });

        request.on('error', (err) => {
            done(new Error(`Request error: ${err.message}`));
        });

        request.setTimeout(15000, () => {
            request.destroy();
            done(new Error('Request timeout'));
        });
    }).then((data) => {
        result = { status: 'success', msg: '', oriData: data };
    }).catch((e) => {
        result.msg = e.message || e;
    });
    return result;
};
```

### C. `gallery.js` — `save` handler 中未声明变量导致全局污染

**文件：** `javascript/blog/gallery.js` 第 98 行

**问题：** `article = {...}` 缺少 `let`/`const`/`var`，会成为全局变量。

```javascript
// 修改前

// 修改后
let article = { "time": time, "auth": auth, "title": title, "text": text, "ablum": ablum };
```
